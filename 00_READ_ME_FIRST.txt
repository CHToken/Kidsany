🚀 QUICK START (5 MINUTES)
═══════════════════════════════════════════════════════════════════

# 1. Database Setup
psql -U postgres
CREATE DATABASE kidsany_db;
\q

# 2. Backend Setup
cd backend
npm install
cp .env.example .env
# Edit .env with your credentials
npm run dev

# 3. Frontend Setup (new terminal)
cd frontend
npm install
npm run dev

# 4. Test
curl http://localhost:5000/health
